if(DESKFLOW_AUDIO_FETCH_DEPENDENCIES)
  include(FetchContent)
  # Do not inherit the parent Qt autogen or CTest options into plain C dependencies.
  set(audio_saved_testing ${BUILD_TESTING})
  set(audio_saved_automoc ${CMAKE_AUTOMOC})
  set(audio_saved_autouic ${CMAKE_AUTOUIC})
  set(audio_saved_autorcc ${CMAKE_AUTORCC})
  set(BUILD_TESTING OFF)
  set(CMAKE_AUTOMOC OFF)
  set(CMAKE_AUTOUIC OFF)
  set(CMAKE_AUTORCC OFF)
  set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(LIBSRTP_TEST_APPS OFF CACHE BOOL "" FORCE)
  set(ENABLE_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
  set(ENABLE_DEBUG_LOGGING OFF CACHE BOOL "" FORCE)
  set(ERR_REPORTING_STDOUT OFF CACHE BOOL "" FORCE)
  set(ERR_REPORTING_FILE "" CACHE FILEPATH "" FORCE)
  FetchContent_Declare(audio_opus
    URL https://codeload.github.com/xiph/opus/tar.gz/refs/tags/v1.5.2
    URL_HASH SHA256=9480e329e989f70d69886ded470c7f8cfe6c0667cc4196d4837ac9e668fb7404)
  FetchContent_Declare(audio_srtp
    URL https://codeload.github.com/cisco/libsrtp/tar.gz/refs/tags/v2.7.0
    URL_HASH SHA256=54facb1727a557c2a76b91194dcb2d0a453aaf8e2d0cbbf1e3c2848c323e28ad)
  FetchContent_MakeAvailable(audio_opus audio_srtp)
  set(BUILD_TESTING ${audio_saved_testing})
  set(CMAKE_AUTOMOC ${audio_saved_automoc})
  set(CMAKE_AUTOUIC ${audio_saved_autouic})
  set(CMAKE_AUTORCC ${audio_saved_autorcc})
  # libSRTP's build-tree include layout differs from its installed srtp2/ layout.
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/srtp-include/srtp2)
  configure_file(${audio_srtp_SOURCE_DIR}/include/srtp.h
    ${CMAKE_CURRENT_BINARY_DIR}/srtp-include/srtp2/srtp.h COPYONLY)
else()
  find_package(Opus CONFIG REQUIRED)
  find_package(libSRTP CONFIG REQUIRED)
endif()

if(DESKFLOW_AUDIO_BUILD_TESTS)
  enable_testing()
  foreach(suite Dsp Transport Codec Engine NetworkClock Resampler RtSafety)
    string(TOLOWER ${suite} lower)
    add_executable(audio-${lower}-tests tests/${suite}Tests.cpp)
    set_target_properties(audio-${lower}-tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(audio-${lower}-tests PRIVATE deskflow-audio)
    if(MSVC)
      target_compile_options(audio-${lower}-tests PRIVATE /W4 /permissive- /utf-8 /wd4324)
    endif()
    add_test(NAME audio-${lower} COMMAND audio-${lower}-tests)
    if(suite STREQUAL "Engine")
      # Deterministic worker/device interleaving tests must fail, not hang CI forever.
      set_tests_properties(audio-${lower} PROPERTIES TIMEOUT 30)
    endif()
    if(suite STREQUAL "RtSafety")
      set_tests_properties(audio-${lower} PROPERTIES SKIP_RETURN_CODE 77)
    endif()
  endforeach()
  if(WIN32)
