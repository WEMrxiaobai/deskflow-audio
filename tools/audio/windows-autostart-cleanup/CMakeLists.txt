cmake_minimum_required(VERSION 3.24)
project(DeskflowAutoStartCleanup LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Windows Installer cleanup requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
# 自包含 MSI 操作不能依赖已被 RemoveFiles 删除的 Qt 或 VC 运行库。
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W4 /WX)
endif()
add_library(autostart-cleanup STATIC AutoStartCleanup.cpp)
target_compile_definitions(autostart-cleanup PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(autostart-cleanup PUBLIC advapi32)
add_library(DeskflowAutoStartCleanup SHARED InstallerAction.cpp)
target_link_libraries(DeskflowAutoStartCleanup PRIVATE autostart-cleanup msi)
include(CTest)
if(BUILD_TESTING)
  add_executable(autostart-cleanup-tests AutoStartCleanupTests.cpp)
  target_link_libraries(autostart-cleanup-tests PRIVATE autostart-cleanup)
  add_test(NAME autostart-cleanup COMMAND autostart-cleanup-tests)
endif()
